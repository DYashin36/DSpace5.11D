SELECT c.collection_id,
       h.handle,
       mv.text_value AS name
FROM collection c
JOIN handle h ON c.collection_id = h.resource_id
LEFT JOIN metadatavalue mv ON (mv.resource_id = c.collection_id AND mv.resource_type_id = 3)
LEFT JOIN metadatafieldregistry mfr ON mv.metadata_field_id = mfr.metadata_field_id
LEFT JOIN metadataschemaregistry msr ON mfr.metadata_schema_id = msr.metadata_schema_id
WHERE h.resource_type_id = 3
  AND msr.short_id = 'dc'
  AND mfr.element = 'title'
ORDER BY c.collection_id;
